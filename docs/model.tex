\begin{aligned}
\textbf{Likelihood:} \\
y_{ij} &\sim \text{Bernoulli}(\frac{\theta_{ij}}{1 - \theta_{ij}}) \\
\theta_{ij} &= \alpha + u_j + (\beta + w_j) \cdot \text{Frequency}_{ij} \\ \\
\textbf{Prior:} \\
\alpha, \beta &\sim \text{Normal}(0, 5) \\
u_j, w_j &\sim \text{Normal}(0, \sigma_{u, w}) \\
\sigma_{u, w} &\sim \text{HalfNormal}(1) \\ \\
\textbf{Where:} \\
&i~\text{indexes each observation and }j~\text{indexes the language it belongs to} \\
&y_{ij}~\text{indexes whether the syllable follows (1) or violates (0) the SSP}\\
&u_j, w_j~\text{are language-specific adjustments to }\alpha\text{ and }\beta \\
\end{aligned}
